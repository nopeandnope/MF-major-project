# src/data_quality_validator.py
import pandas as pd
import numpy as np
from datetime import datetime, timedelta

class DataQualityValidator:
    """Validate and ensure data quality for mutual fund NAV data"""

    def __init__(self, data_source='mfapi.in'):
        self.data_source = data_source
        self.quality_report = {}

    def validate_data_quality(self, df, scheme_code):
        """Comprehensive data quality checks"""

        results = {
            'scheme_code': scheme_code,
            'total_records': len(df),
            'date_range': f"{df['date'].min()} to {df['date'].max()}",
            'checks': {}
        }

        # 1. Check for missing values
        missing_pct = (df.isnull().sum() / len(df) * 100).to_dict()
        results['checks']['missing_values'] = missing_pct

        # 2. Check for outliers (>50% daily change likely error)
        df['daily_change'] = df['nav'].pct_change()
        outliers = df[abs(df['daily_change']) > 0.5]
        results['checks']['outliers'] = len(outliers)

        # 3. Verify date continuity (check for gaps > 5 days)
        df['date'] = pd.to_datetime(df['date'])
        df = df.sort_values('date')
        df['date_diff'] = df['date'].diff().dt.days
        gaps = df[df['date_diff'] > 5]
        results['checks']['date_gaps'] = len(gaps)

        # 4. Check for duplicate dates
        duplicates = df[df.duplicated(subset=['date'], keep=False)]
        results['checks']['duplicate_dates'] = len(duplicates)

        # 5. Validate NAV values (should be positive)
        invalid_nav = df[df['nav'] <= 0]
        results['checks']['invalid_nav_values'] = len(invalid_nav)

        # 6. Check data freshness (last update within 3 days?)
        last_date = df['date'].max()
        days_since_update = (datetime.now() - last_date).days
        results['checks']['days_since_last_update'] = days_since_update

        # Overall quality score (0-100)
        score = 100
        if missing_pct.get('nav', 0) > 1: score -= 20
        if len(outliers) > 0: score -= 15
        if len(gaps) > 10: score -= 15
        if len(duplicates) > 0: score -= 10
        if len(invalid_nav) > 0: score -= 20
        if days_since_update > 3: score -= 10

        results['quality_score'] = max(0, score)
        results['status'] = 'PASS' if score >= 70 else 'FAIL'

        return results

    def cross_validate_with_backup(self, primary_data, backup_data):
        """Cross-validate data from multiple sources"""

        # Merge on date
        merged = pd.merge(primary_data, backup_data,
                         on='date', suffixes=('_primary', '_backup'))

        # Calculate difference
        merged['nav_diff'] = abs(merged['nav_primary'] - merged['nav_backup'])
        merged['nav_diff_pct'] = (merged['nav_diff'] / merged['nav_primary']) * 100

        # Flag discrepancies (>1% difference)
        discrepancies = merged[merged['nav_diff_pct'] > 1]

        return {
            'total_compared': len(merged),
            'discrepancies': len(discrepancies),
            'avg_difference_pct': merged['nav_diff_pct'].mean(),
            'max_difference_pct': merged['nav_diff_pct'].max(),
            'data_agreement': (1 - len(discrepancies)/len(merged)) * 100
        }

    def generate_data_metadata(self, df):
        """Generate metadata for dataset versioning"""

        return {
            'collection_date': datetime.now().isoformat(),
            'data_source': self.data_source,
            'num_funds': df['scheme_code'].nunique(),
            'total_records': len(df),
            'date_range': {
                'start': df['date'].min().isoformat(),
                'end': df['date'].max().isoformat(),
                'years': (df['date'].max() - df['date'].min()).days / 365.25
            },
            'data_version': 'v1.0',
            'quality_score': self.validate_data_quality(df, 'overall')['quality_score']
        }

- Easier learning curve than Backtrader
- Active development and community
- Pythonic, intuitive API

# backtesting/mf_backtest_engine.py
import bt
import pandas as pd
import numpy as np
import quantstats as qs
from datetime import datetime, timedelta

class MutualFundBacktester:
    """
    Comprehensive backtesting engine for mutual fund strategies
    """

    def __init__(self, historical_data_path, benchmark_data_path=None):
        """
        Initialize backtester

        Args:
            historical_data_path: Path to historical NAV data (CSV)
            benchmark_data_path: Path to benchmark data (e.g., Nifty 50)
        """
        self.data = self.load_data(historical_data_path)
        self.benchmark = self.load_benchmark(benchmark_data_path)
        self.results = {}

    def load_data(self, path):
        """Load and prepare historical NAV data"""
        df = pd.read_csv(path, parse_dates=['date'])

        # Pivot to get funds as columns
        pivot_df = df.pivot(index='date', columns='scheme_name', values='nav')

        # Calculate returns
        returns_df = pivot_df.pct_change().fillna(0)

        return {
            'prices': pivot_df,
            'returns': returns_df
        }

    def load_benchmark(self, path):
        """Load benchmark data (Nifty 50)"""
        if path is None:
            return None
        df = pd.read_csv(path, parse_dates=['date'], index_col='date')
        return df['close']

    def create_equal_weight_strategy(self, funds_list, rebalance_freq='quarterly'):
        """
        Equal weight strategy with periodic rebalancing

        Args:
            funds_list: List of fund names to include
            rebalance_freq: 'monthly', 'quarterly', 'yearly'
        """
        # Select data for chosen funds
        data = self.data['prices'][funds_list]

        # Define rebalancing frequency
        rebalance_algos = {
            'monthly': bt.algos.RunMonthly(),
            'quarterly': bt.algos.RunQuarterly(),
            'yearly': bt.algos.RunYearly()
        }

        # Create strategy
        strategy = bt.Strategy(
            f'EqualWeight_{rebalance_freq}',
            [
                rebalance_algos[rebalance_freq],
                bt.algos.SelectAll(),
                bt.algos.WeighEqually(),
                bt.algos.Rebalance()
            ]
        )

        return bt.Backtest(strategy, data)

    def create_model_based_strategy(self, model_predictions, rebalance_freq='quarterly'):
        """
        Strategy based on AI model predictions

        Args:
            model_predictions: DataFrame with predicted returns/scores
            rebalance_freq: Rebalancing frequency
        """
        # Custom algo to weight based on model predictions
        class WeighByModelPrediction(bt.Algo):
            def __init__(self, predictions):
                super(WeighByModelPrediction, self).__init__()
                self.predictions = predictions

            def __call__(self, target):
                # Get predictions for current date
                current_date = target.now
                if current_date not in self.predictions.index:
                    return True

                pred = self.predictions.loc[current_date]

                # Normalize to weights (only positive predictions)
                pred_positive = pred[pred > 0]
                if len(pred_positive) == 0:
                    return True

                weights = pred_positive / pred_positive.sum()
                target.temp['weights'] = weights

                return True

        rebalance_algos = {
            'monthly': bt.algos.RunMonthly(),
            'quarterly': bt.algos.RunQuarterly(),
            'yearly': bt.algos.RunYearly()
        }

        strategy = bt.Strategy(
            f'ModelBased_{rebalance_freq}',
            [
                rebalance_algos[rebalance_freq],
                WeighByModelPrediction(model_predictions),
                bt.algos.Rebalance()
            ]
        )

        data = self.data['prices'][model_predictions.columns]
        return bt.Backtest(strategy, data)

    def run_timeframe_tests(self, strategy_name='equal_weight'):
        """
        Run backtests across multiple timeframes
        """
        results = {}

        # Define timeframes (end date is most recent)
        end_date = self.data['prices'].index.max()

        timeframes = {
            '3_months': end_date - timedelta(days=90),
            '6_months': end_date - timedelta(days=180),
            '1_year': end_date - timedelta(days=365),
            '3_years': end_date - timedelta(days=365*3),
            '5_years': end_date - timedelta(days=365*5),
            '10_years': end_date - timedelta(days=365*10)
        }

        # Select top 10 funds (example)
        top_funds = self.data['returns'].mean().nlargest(10).index.tolist()

        for period_name, start_date in timeframes.items():
            # Filter data for this timeframe
            period_data = self.data['prices'][
                (self.data['prices'].index >= start_date) &
                (self.data['prices'].index <= end_date)
            ]

            if len(period_data) < 30:  # Skip if insufficient data
                continue

            # Create and run backtest
            backtest = self.create_equal_weight_strategy(
                top_funds,
                rebalance_freq='quarterly'
            )

            result = bt.run(backtest)
            results[period_name] = result

        return results

    def test_crisis_scenarios(self):
        """
        Test strategy performance during market crashes
        """
        crisis_periods = {
            'covid_crash_and_recovery': {
                'start': '2019-01-01',
                'end': '2021-12-31',
                'description': 'COVID-19 crash (Feb-Mar 2020) and recovery'
            },
            'covid_crash_only': {
                'start': '2020-01-01',
                'end': '2020-06-30',
                'description': 'COVID-19 crash period only'
            },
            '2022_correction': {
                'start': '2022-01-01',
                'end': '2022-12-31',
                'description': '2022 market correction'
            },
            'demonetization': {
                'start': '2016-09-01',
                'end': '2017-03-31',
                'description': '2016 demonetization impact'
            }
        }

        results = {}
        top_funds = self.data['returns'].mean().nlargest(10).index.tolist()

        for crisis_name, period_info in crisis_periods.items():
            # Filter data
            crisis_data = self.data['prices'][
                (self.data['prices'].index >= period_info['start']) &
                (self.data['prices'].index <= period_info['end'])
            ]

            if len(crisis_data) < 30:
                continue

            # Run backtest
            backtest = self.create_equal_weight_strategy(
                top_funds,
                rebalance_freq='monthly'  # Monthly rebalancing during crisis
            )

            result = bt.run(backtest)
            results[crisis_name] = {
                'result': result,
                'description': period_info['description']
            }

        return results

    def calculate_performance_metrics(self, backtest_result):
        """
        Calculate comprehensive performance metrics
        """
        returns = backtest_result.returns

        # Basic metrics
        total_return = (backtest_result.prices.iloc[-1] /
                       backtest_result.prices.iloc[0] - 1) * 100

        # Annualized return
        years = (backtest_result.prices.index[-1] -
                backtest_result.prices.index[0]).days / 365.25
        annualized_return = ((1 + total_return/100) ** (1/years) - 1) * 100

        # Volatility (annualized)
        volatility = returns.std() * np.sqrt(252) * 100

        # Sharpe Ratio (assuming 6.5% risk-free rate for India)
        risk_free_rate = 0.065
        sharpe_ratio = (annualized_return/100 - risk_free_rate) / (volatility/100)

        # Maximum Drawdown
        cumulative = (1 + returns).cumprod()
        running_max = cumulative.expanding().max()
        drawdown = (cumulative - running_max) / running_max
        max_drawdown = drawdown.min() * 100

        # Sortino Ratio (downside deviation)
        downside_returns = returns[returns < 0]
        downside_std = downside_returns.std() * np.sqrt(252)
        sortino_ratio = (annualized_return/100 - risk_free_rate) / downside_std

        # Calmar Ratio
        calmar_ratio = (annualized_return/100) / abs(max_drawdown/100)

        # Win rate
        win_rate = (returns > 0).sum() / len(returns) * 100

        return {
            'total_return': round(total_return, 2),
            'annualized_return': round(annualized_return, 2),
            'volatility': round(volatility, 2),
            'sharpe_ratio': round(sharpe_ratio, 2),
            'max_drawdown': round(max_drawdown, 2),
            'sortino_ratio': round(sortino_ratio, 2),
            'calmar_ratio': round(calmar_ratio, 2),
            'win_rate': round(win_rate, 2),
            'total_trades': len(returns),
            'start_date': backtest_result.prices.index[0],
            'end_date': backtest_result.prices.index[-1]
        }

    def generate_html_report(self, backtest_result, output_path='backtest_report.html'):
        """
        Generate comprehensive HTML report using quantstats
        """
        returns = backtest_result.returns

        if self.benchmark is not None:
            qs.reports.html(returns, self.benchmark, output=output_path)
        else:
            qs.reports.html(returns, output=output_path)

        print(f"Report generated: {output_path}")

    def compare_strategies(self, strategies_dict):
        """
        Compare multiple strategies side-by-side

        Args:
            strategies_dict: Dict of {strategy_name: backtest_result}
        """
        comparison = {}

        for name, result in strategies_dict.items():
            metrics = self.calculate_performance_metrics(result)
            comparison[name] = metrics

        # Create comparison DataFrame
        comparison_df = pd.DataFrame(comparison).T

        # Highlight best performers
        comparison_df['rank_sharpe'] = comparison_df['sharpe_ratio'].rank(ascending=False)
        comparison_df['rank_return'] = comparison_df['annualized_return'].rank(ascending=False)
        comparison_df['rank_drawdown'] = comparison_df['max_drawdown'].rank(ascending=True)

        return comparison_df

# src/advanced_training_data_generator.py
import pandas as pd
import numpy as np
from datetime import datetime, timedelta
import json

class AdvancedTrainingDataGenerator:
    """
    Generate high-quality training data for financial LLM
    """

    def __init__(self, historical_db_path):
        self.historical_data = pd.read_csv(historical_db_path)
        self.nifty_data = None  # Load separately
        self.vix_data = None

    def calculate_returns(self, nav_series, periods=[1, 7, 30, 90, 365]):
        """Calculate returns for multiple periods"""
        returns = {}
        for period in periods:
            if len(nav_series) > period:
                returns[f'{period}day'] = (
                    (nav_series.iloc[-1] / nav_series.iloc[-period] - 1) * 100
                )
        return returns

    def calculate_risk_metrics(self, returns_series):
        """Calculate volatility, Sharpe, etc."""
        volatility = returns_series.std() * np.sqrt(252) * 100
        sharpe = (returns_series.mean() * 252 - 0.065) / (volatility / 100)
        max_dd = ((returns_series.cumsum() -
                  returns_series.cumsum().expanding().max()).min())

        return {
            'volatility': round(volatility, 2),
            'sharpe': round(sharpe, 2),
            'max_drawdown': round(max_dd * 100, 2)
        }

    def determine_market_phase(self, nifty_returns):
        """Classify market as bull/bear/sideways"""
        if nifty_returns['30day'] > 5:
            return 'Bull Market'
        elif nifty_returns['30day'] < -5:
            return 'Bear Market'
        else:
            return 'Sideways Market'

    def find_similar_historical_pattern(self, current_conditions):
        """Find similar market conditions in history"""
        # Machine learning approach to find similar patterns
        # (simplified version here)
        return {
            'period': 'Q2 2019',
            'similarity': 0.85,
            'outcome': '+18% over 6 months'
        }

    def generate_training_example(self, scheme_code, as_of_date):
        """
        Generate single rich training example
        """
        # Get fund data
        fund_data = self.historical_data[
            (self.historical_data['scheme_code'] == scheme_code) &
            (self.historical_data['date'] <= as_of_date)
        ].sort_values('date')

        if len(fund_data) < 365:  # Need at least 1 year
            return None

        # Calculate metrics
        nav_series = fund_data['nav']
        returns = self.calculate_returns(nav_series)
        risk_metrics = self.calculate_risk_metrics(
            fund_data.set_index('date')['nav'].pct_change()
        )

        # Market conditions
        nifty_returns = self.calculate_returns(self.nifty_data['close'])
        market_phase = self.determine_market_phase(nifty_returns)

        # Generate prompt
        prompt = f"""[CONTEXT]
Fund Name: {fund_data.iloc[-1]['scheme_name']}
Scheme Code: {scheme_code}
Category: {fund_data.iloc[-1]['category']}
Current NAV: {nav_series.iloc[-1]:.2f}

[RECENT PERFORMANCE]
30-day Return: {returns.get('30day', 0):.2f}%
90-day Return: {returns.get('90day', 0):.2f}%
1-year Return: {returns.get('365day', 0):.2f}%

[RISK METRICS]
Volatility (30-day): {risk_metrics['volatility']:.2f}%
Sharpe Ratio: {risk_metrics['sharpe']:.2f}
Maximum Drawdown: {risk_metrics['max_drawdown']:.2f}%

[MARKET CONDITIONS]
Market Phase: {market_phase}
Nifty 50 (30-day): {nifty_returns.get('30day', 0):.2f}%
India VIX: {self.vix_data.iloc[-1]['vix']:.2f}

[QUESTION]
Predict 3-month expected return and provide recommendation.

[ANSWER]
"""

        # Generate answer based on actual future performance
        future_data = self.historical_data[
            (self.historical_data['scheme_code'] == scheme_code) &
            (self.historical_data['date'] > as_of_date) &
            (self.historical_data['date'] <= as_of_date + timedelta(days=90))
        ]

        if len(future_data) > 0:
            actual_return = (
                (future_data.iloc[-1]['nav'] / nav_series.iloc[-1] - 1) * 100
            )

            # Generate recommendation
            if actual_return > 8:
                recommendation = "STRONG BUY"
            elif actual_return > 4:
                recommendation = "BUY"
            elif actual_return > -2:
                recommendation = "HOLD"
            else:
                recommendation = "SELL"

            answer = f"""Expected 3-month return: {actual_return:.2f}%
Recommendation: {recommendation}
Risk level: {'High' if risk_metrics['volatility'] > 20 else 'Moderate' if risk_metrics['volatility'] > 12 else 'Low'}

Reasoning:
1. Recent momentum: {returns.get('90day', 0):.1f}% in last 90 days
2. Market phase: {market_phase}
3. Risk-adjusted returns (Sharpe): {risk_metrics['sharpe']:.2f}
4. Volatility: {risk_metrics['volatility']:.1f}%
"""

            return {
                'prompt': prompt,
                'completion': answer,
                'metadata': {
                    'scheme_code': scheme_code,
                    'as_of_date': as_of_date,
                    'actual_return': actual_return
                }
            }

        return None

    def generate_dataset(self, num_examples=10000):
        """Generate complete training dataset"""

        examples = []
        unique_funds = self.historical_data['scheme_code'].unique()

        for scheme_code in unique_funds:
            fund_dates = self.historical_data[
                self.historical_data['scheme_code'] == scheme_code
            ]['date'].unique()

            # Sample random dates for this fund
            sample_dates = np.random.choice(
                fund_dates[:-90],  # Exclude last 90 days
                size=min(50, len(fund_dates) - 90),
                replace=False
            )

            for date in sample_dates:
                example = self.generate_training_example(scheme_code, date)
                if example:
                    examples.append(example)

                if len(examples) >= num_examples:
                    break

            if len(examples) >= num_examples:
                break

        return examples

    def save_dataset(self, examples, output_path='training_data.jsonl'):
        """Save in JSONL format for training"""
        with open(output_path, 'w') as f:
            for example in examples:
                json.dump(example, f)
                f.write('\n')

# evaluation/model_evaluator.py
import numpy as np
import pandas as pd
from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score
from scipy.stats import spearmanr
import matplotlib.pyplot as plt

class ModelPerformanceEvaluator:
    """
    Comprehensive evaluation of model predictions
    """

    def __init__(self):
        self.risk_free_rate = 0.065  # India government bonds
        self.predictions = []
        self.actuals = []

    def add_prediction(self, predicted, actual, metadata=None):
        """Add prediction-actual pair"""
        self.predictions.append({
            'predicted': predicted,
            'actual': actual,
            'metadata': metadata or {}
        })

    def calculate_prediction_metrics(self):
        """Calculate accuracy metrics"""

        y_pred = np.array([p['predicted'] for p in self.predictions])
        y_true = np.array([p['actual'] for p in self.predictions])

        # Root Mean Squared Error
        rmse = np.sqrt(mean_squared_error(y_true, y_pred))

        # Mean Absolute Error
        mae = mean_absolute_error(y_true, y_pred)

        # Mean Absolute Percentage Error
        mape = np.mean(np.abs((y_true - y_pred) / y_true)) * 100

        # R² Score
        r2 = r2_score(y_true, y_pred)

        # Directional Accuracy
        directional_accuracy = self.calculate_directional_accuracy(y_pred, y_true)

        # Spearman Correlation (ranking)
        spearman_corr, _ = spearmanr(y_true, y_pred)

        return {
            'rmse': round(rmse, 4),
            'mae': round(mae, 4),
            'mape': round(mape, 2),
            'r2_score': round(r2, 4),
            'directional_accuracy': round(directional_accuracy, 2),
            'spearman_correlation': round(spearman_corr, 4)
        }

    def calculate_directional_accuracy(self, y_pred, y_true):
        """
        Percentage of times model correctly predicted direction
        (positive vs negative return)
        """
        pred_direction = np.sign(y_pred)
        true_direction = np.sign(y_true)

        correct = np.sum(pred_direction == true_direction)
        total = len(y_pred)

        return (correct / total) * 100

    def calculate_sharpe_ratio(self, returns):
        """Calculate Sharpe ratio of portfolio based on predictions"""
        excess_returns = returns - self.risk_free_rate
        sharpe = (np.mean(excess_returns) / np.std(excess_returns)) * np.sqrt(252)
        return sharpe

    def calculate_information_ratio(self, returns, benchmark_returns):
        """Information ratio vs benchmark"""
        excess_returns = returns - benchmark_returns
        tracking_error = np.std(excess_returns)

        if tracking_error == 0:
            return 0

        return np.mean(excess_returns) / tracking_error * np.sqrt(252)

    def backtest_on_predictions(self):
        """
        Simulate trading based on model predictions
        """
        # Sort predictions by confidence/expected return
        sorted_preds = sorted(
            self.predictions,
            key=lambda x: x['predicted'],
            reverse=True
        )

        # Select top 20% (quintile analysis)
        top_quintile = sorted_preds[:len(sorted_preds)//5]
        top_returns = [p['actual'] for p in top_quintile]

        # Bottom quintile
        bottom_quintile = sorted_preds[-len(sorted_preds)//5:]
        bottom_returns = [p['actual'] for p in bottom_quintile]

        return {
            'top_quintile_avg_return': np.mean(top_returns),
            'bottom_quintile_avg_return': np.mean(bottom_returns),
            'quintile_spread': np.mean(top_returns) - np.mean(bottom_returns),
            'top_quintile_sharpe': self.calculate_sharpe_ratio(
                np.array(top_returns)/100
            )
        }

    def generate_evaluation_report(self):
        """Comprehensive evaluation report"""

        print("=" * 70)
        print("MODEL PERFORMANCE EVALUATION REPORT")
        print("=" * 70)

        # Prediction Metrics
        metrics = self.calculate_prediction_metrics()
        print("\n1. PREDICTION ACCURACY METRICS")
        print("-" * 70)
        print(f"RMSE:                    {metrics['rmse']:.4f}")
        print(f"MAE:                     {metrics['mae']:.4f}")
        print(f"MAPE:                    {metrics['mape']:.2f}%")
        print(f"R² Score:                {metrics['r2_score']:.4f}")
        print(f"Directional Accuracy:    {metrics['directional_accuracy']:.2f}%")
        print(f"Spearman Correlation:    {metrics['spearman_correlation']:.4f}")

        # Performance benchmarks
        print("\n2. PERFORMANCE EVALUATION")
        print("-" * 70)
        if metrics['rmse'] < 5:
            print("✓ RMSE: EXCELLENT (< 5%)")
        elif metrics['rmse'] < 8:
            print("✓ RMSE: GOOD (< 8%)")
        else:
            print("✗ RMSE: NEEDS IMPROVEMENT (> 8%)")

        if metrics['directional_accuracy'] > 65:
            print("✓ Directional Accuracy: EXCELLENT (> 65%)")
        elif metrics['directional_accuracy'] > 55:
            print("✓ Directional Accuracy: GOOD (> 55%)")
        else:
            print("✗ Directional Accuracy: NEEDS IMPROVEMENT (< 55%)")

        # Quintile Analysis
        quintile_results = self.backtest_on_predictions()
        print("\n3. QUINTILE ANALYSIS (Top 20% vs Bottom 20%)")
        print("-" * 70)
        print(f"Top Quintile Avg Return:    {quintile_results['top_quintile_avg_return']:.2f}%")
        print(f"Bottom Quintile Avg Return: {quintile_results['bottom_quintile_avg_return']:.2f}%")
        print(f"Spread:                     {quintile_results['quintile_spread']:.2f}%")
        print(f"Top Quintile Sharpe:        {quintile_results['top_quintile_sharpe']:.2f}")

        print("\n" + "=" * 70)

        return {
            'accuracy_metrics': metrics,
            'quintile_analysis': quintile_results
        }

from sklearn.model_selection import TimeSeriesSplit

    train_data = data.iloc[train_idx]
    test_data = data.iloc[test_idx]

    # Train model on train_data
    model.fit(train_data)

    # Evaluate on test_data
    predictions = model.predict(test_data)
    metrics = calculate_metrics(predictions, test_data['actual'])

# Average metrics across folds

# news_portal/news_aggregator.py
import requests
from datetime import datetime, timedelta
from textblob import TextBlob
from apscheduler.schedulers.background import BackgroundScheduler
import sqlite3
import json

class NewsAggregator:
    """
    Aggregate news from multiple sources with sentiment analysis
    """

    def __init__(self, db_path='data/news.db'):
        self.db_path = db_path
        self.alpha_vantage_key = "YOUR_KEY"
        self.newsapi_key = "YOUR_KEY"
        self.scheduler = BackgroundScheduler()
        self.init_database()

    def init_database(self):
        """Initialize news database"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        cursor.execute('''
            CREATE TABLE IF NOT EXISTS news_articles (
                article_id INTEGER PRIMARY KEY AUTOINCREMENT,
                title TEXT NOT NULL,
                url TEXT UNIQUE NOT NULL,
                source TEXT,
                published_at TIMESTAMP,
                content TEXT,
                sentiment TEXT,
                sentiment_score REAL,
                mentioned_funds TEXT,
                relevance_score REAL,
                category TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')

        cursor.execute('''
            CREATE TABLE IF NOT EXISTS market_performance (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                date DATE UNIQUE,
                nifty_50_close REAL,
                nifty_50_change REAL,
                india_vix REAL,
                top_performers TEXT,
                top_losers TEXT,
                market_sentiment TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')

        conn.commit()
        conn.close()

    def fetch_alpha_vantage_news(self):
        """Fetch from Alpha Vantage"""
        url = "https://www.alphavantage.co/query"
        params = {
            'function': 'NEWS_SENTIMENT',
            'topics': 'financial_markets,economy_fiscal',
            'tickers': 'NSE:NIFTY50',
            'apikey': self.alpha_vantage_key,
            'limit': 50
        }

        try:
            response = requests.get(url, params=params, timeout=10)
            data = response.json()

            articles = []
            for item in data.get('feed', []):
                articles.append({
                    'title': item.get('title'),
                    'url': item.get('url'),
                    'source': item.get('source'),
                    'published_at': item.get('time_published'),
                    'content': item.get('summary'),
                    'sentiment': item.get('overall_sentiment_label'),
                    'sentiment_score': item.get('overall_sentiment_score'),
                    'category': 'market_news'
                })

            return articles
        except Exception as e:
            print(f"Error fetching from Alpha Vantage: {e}")
            return []

    def fetch_newsapi_articles(self):
        """Fetch from NewsAPI"""
        url = "https://newsapi.org/v2/everything"
        params = {
            'q': 'mutual fund India OR SEBI OR investment',
            'language': 'en',
            'sortBy': 'publishedAt',
            'apiKey': self.newsapi_key,
            'pageSize': 50
        }

        try:
            response = requests.get(url, params=params, timeout=10)
            data = response.json()

            articles = []
            for item in data.get('articles', []):
                articles.append({
                    'title': item.get('title'),
                    'url': item.get('url'),
                    'source': item.get('source', {}).get('name'),
                    'published_at': item.get('publishedAt'),
                    'content': item.get('description'),
                    'category': 'mutual_fund_news'
                })

            return articles
        except Exception as e:
            print(f"Error fetching from NewsAPI: {e}")
            return []

    def analyze_sentiment(self, text):
        """Analyze sentiment using TextBlob"""
        if not text:
            return 'neutral', 0.0

        analysis = TextBlob(text)
        polarity = analysis.sentiment.polarity

        if polarity > 0.1:
            sentiment = 'positive'
        elif polarity < -0.1:
            sentiment = 'negative'
        else:
            sentiment = 'neutral'

        return sentiment, polarity

    def extract_fund_mentions(self, text):
        """Extract mentioned fund names from article"""
        # Load list of fund names from database
        # Simple keyword matching (can be improved with NER)

        fund_keywords = [
            'HDFC', 'SBI', 'ICICI', 'Axis', 'Aditya Birla',
            'UTI', 'Mirae Asset', 'Nippon India', 'Kotak'
        ]

        mentioned = []
        text_lower = text.lower()

        for keyword in fund_keywords:
            if keyword.lower() in text_lower:
                mentioned.append(keyword)

        return mentioned

    def calculate_relevance_score(self, article):
        """Score article relevance (0-100)"""
        score = 0

        # Title mentions mutual fund (+30)
        if 'mutual fund' in article['title'].lower():
            score += 30

        # Content mentions funds (+20)
        if article.get('content') and 'mutual fund' in article['content'].lower():
            score += 20

        # Recent article (+20)
        pub_date = datetime.fromisoformat(article['published_at'].replace('Z', ''))
        hours_old = (datetime.now() - pub_date).total_seconds() / 3600
        if hours_old < 24:
            score += 20
        elif hours_old < 72:
            score += 10

        # Mentions specific funds (+15)
        if article.get('mentioned_funds'):
            score += 15

        # From trusted source (+15)
        trusted_sources = ['Economic Times', 'MoneyControl', 'Value Research']
        if any(source in article.get('source', '') for source in trusted_sources):
            score += 15

        return min(score, 100)

    def fetch_and_process_news(self):
        """Main aggregation function"""
        print(f"[{datetime.now()}] Fetching news...")

        # Fetch from all sources
        news = []
        news.extend(self.fetch_alpha_vantage_news())
        news.extend(self.fetch_newsapi_articles())

        # Process each article
        processed = []
        for article in news:
            # Analyze sentiment if not provided
            if not article.get('sentiment'):
                sentiment, score = self.analyze_sentiment(
                    article.get('content', '') or article.get('title', '')
                )
                article['sentiment'] = sentiment
                article['sentiment_score'] = score

            # Extract fund mentions
            article['mentioned_funds'] = json.dumps(
                self.extract_fund_mentions(article.get('content', '') or article.get('title', ''))
            )

            # Calculate relevance
            article['relevance_score'] = self.calculate_relevance_score(article)

            processed.append(article)

        # Save to database
        self.save_articles(processed)

        print(f"[{datetime.now()}] Processed {len(processed)} articles")

    def save_articles(self, articles):
        """Save articles to database"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        for article in articles:
            try:
                cursor.execute('''
                    INSERT OR IGNORE INTO news_articles
                    (title, url, source, published_at, content, sentiment,
                     sentiment_score, mentioned_funds, relevance_score, category)
                    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                ''', (
                    article.get('title'),
                    article.get('url'),
                    article.get('source'),
                    article.get('published_at'),
                    article.get('content'),
                    article.get('sentiment'),
                    article.get('sentiment_score'),
                    article.get('mentioned_funds'),
                    article.get('relevance_score'),
                    article.get('category')
                ))
            except Exception as e:
                print(f"Error saving article: {e}")

        conn.commit()
        conn.close()

    def get_latest_news(self, limit=20, min_relevance=50):
        """Retrieve latest news from database"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        cursor.execute('''
            SELECT title, url, source, published_at, sentiment,
                   relevance_score, mentioned_funds
            FROM news_articles
            WHERE relevance_score >= ?
            ORDER BY published_at DESC
            LIMIT ?
        ''', (min_relevance, limit))

        articles = cursor.fetchall()
        conn.close()

        return [
            {
                'title': row[0],
                'url': row[1],
                'source': row[2],
                'published_at': row[3],
                'sentiment': row[4],
                'relevance_score': row[5],
                'mentioned_funds': json.loads(row[6]) if row[6] else []
            }
            for row in articles
        ]

    def start_scheduler(self):
        """Start automated news fetching"""
        # Fetch news every hour
        self.scheduler.add_job(
            self.fetch_and_process_news,
            'interval',
            hours=1,
            id='news_fetch_job'
        )

        self.scheduler.start()
        print("News aggregation scheduler started (runs every hour)")

def verify_password(plain_password: str, hashed_password: str) -> bool:
    """Verify password against hash"""
    return pwd_context.verify(plain_password, hashed_password)

def get_password_hash(password: str) -> str:
    """Hash password using Argon2"""
    return pwd_context.hash(password)

def create_access_token(data: dict) -> str:
    """Create JWT access token (15 minutes)"""
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire, "type": "access"})
    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

def create_refresh_token(data: dict) -> str:
    """Create JWT refresh token (7 days)"""
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(days=REFRESH_TOKEN_EXPIRE_DAYS)
    to_encode.update({"exp": expire, "type": "refresh"})
    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

async def get_current_user(token: str = Depends(oauth2_scheme)):
    """Get current authenticated user from token"""
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )

    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        email: str = payload.get("sub")
        token_type: str = payload.get("type")

        if email is None or token_type != "access":
            raise credentials_exception

        token_data = TokenData(email=email)
    except JWTError:
        raise credentials_exception

    # Fetch user from database
    user = get_user_by_email(token_data.email)  # Implement this
    if user is None:
        raise credentials_exception

    return user

    db: Session = Depends(database.get_db)

    """Login user and return JWT tokens"""

    # Authenticate user
    user = db.query(User).filter(User.email == form_data.username).first()

    if not user or not verify_password(form_data.password, user.password_hash):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Incorrect email or password",
            headers={"WWW-Authenticate": "Bearer"},
        )

    if not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Account is inactive"
        )

    # Create tokens
    access_token = create_access_token(data={"sub": user.email})
    refresh_token = create_refresh_token(data={"sub": user.email})

    # Update last login
    user.last_login = datetime.utcnow()
    db.commit()

    return {
        "access_token": access_token,
        "refresh_token": refresh_token,
        "token_type": "bearer"
    }

# portfolio/portfolio_manager.py
from sqlalchemy.orm import Session
from decimal import Decimal
import requests
from datetime import datetime

class PortfolioManager:
    """Manage user portfolios and holdings"""

    def __init__(self, db: Session):
        self.db = db
        self.mf_api = "https://api.mfapi.in/mf"

    def create_portfolio(self, user_id: int, name: str, description: str = ""):
        """Create new portfolio for user"""
        portfolio = Portfolio(
            user_id=user_id,
            portfolio_name=name,
            description=description
        )
        self.db.add(portfolio)
        self.db.commit()
        return portfolio

    def add_holding(self, portfolio_id: int, scheme_code: str,
                   units: Decimal, purchase_nav: Decimal,
                   purchase_date: datetime):
        """Add or update holding in portfolio"""

        # Fetch fund details
        fund_data = self.get_fund_details(scheme_code)

        # Check if holding exists
        existing = self.db.query(Holding).filter(
            Holding.portfolio_id == portfolio_id,
            Holding.scheme_code == scheme_code
        ).first()

        if existing:
            # Update existing holding (average NAV)
            total_units = existing.units + units
            total_invested = (existing.units * existing.avg_purchase_nav +
                            units * purchase_nav)
            new_avg_nav = total_invested / total_units

            existing.units = total_units
            existing.avg_purchase_nav = new_avg_nav
            existing.invested_amount = total_invested
            existing.last_purchase_date = purchase_date

            holding = existing
        else:
            # Create new holding
            holding = Holding(
                portfolio_id=portfolio_id,
                scheme_code=scheme_code,
                fund_name=fund_data['scheme_name'],
                category=fund_data['category'],
                units=units,
                avg_purchase_nav=purchase_nav,
                invested_amount=units * purchase_nav,
                first_purchase_date=purchase_date,
                last_purchase_date=purchase_date
            )
            self.db.add(holding)

        # Add transaction
        transaction = Transaction(
            holding_id=holding.holding_id,
            portfolio_id=portfolio_id,
            transaction_type='BUY',
            transaction_date=purchase_date,
            units=units,
            nav=purchase_nav,
            amount=units * purchase_nav
        )
        self.db.add(transaction)

        self.db.commit()
        self.update_portfolio_value(portfolio_id)

        return holding

    def get_fund_details(self, scheme_code: str):
        """Fetch fund details from API"""
        response = requests.get(f"{self.mf_api}/{scheme_code}")
        data = response.json()

        return {
            'scheme_name': data['meta']['scheme_name'],
            'category': data['meta'].get('scheme_category', 'Unknown'),
            'current_nav': float(data['data'][0]['nav'])
        }

    def update_portfolio_value(self, portfolio_id: int):
        """Update current value of all holdings and portfolio"""

        holdings = self.db.query(Holding).filter(
            Holding.portfolio_id == portfolio_id
        ).all()

        total_invested = Decimal(0)
        total_current = Decimal(0)

        for holding in holdings:
            # Fetch current NAV
            fund_data = self.get_fund_details(holding.scheme_code)
            current_nav = Decimal(str(fund_data['current_nav']))

            # Update holding
            holding.current_nav = current_nav
            holding.current_value = holding.units * current_nav
            holding.absolute_return = holding.current_value - holding.invested_amount
            holding.return_percentage = (
                (holding.current_value / holding.invested_amount - 1) * 100
            )
            holding.last_updated = datetime.now()

            total_invested += holding.invested_amount
            total_current += holding.current_value

        # Update portfolio
        portfolio = self.db.query(Portfolio).get(portfolio_id)
        portfolio.total_invested = total_invested
        portfolio.current_value = total_current
        portfolio.absolute_return = total_current - total_invested
        portfolio.return_percentage = (
            (total_current / total_invested - 1) * 100 if total_invested > 0 else 0
        )
        portfolio.last_updated = datetime.now()

        self.db.commit()

    def get_portfolio_summary(self, portfolio_id: int):
        """Get comprehensive portfolio summary"""

        portfolio = self.db.query(Portfolio).get(portfolio_id)
        holdings = self.db.query(Holding).filter(
            Holding.portfolio_id == portfolio_id
        ).all()

        # Calculate allocation by category
        category_allocation = {}
        for holding in holdings:
            category = holding.category
            if category not in category_allocation:
                category_allocation[category] = Decimal(0)
            category_allocation[category] += holding.current_value

        # Convert to percentages
        total_value = portfolio.current_value
        category_allocation_pct = {
            category: float((value / total_value) * 100)
            for category, value in category_allocation.items()
        }

        return {
            'portfolio': {
                'name': portfolio.portfolio_name,
                'total_invested': float(portfolio.total_invested),
                'current_value': float(portfolio.current_value),
                'absolute_return': float(portfolio.absolute_return),
                'return_percentage': float(portfolio.return_percentage),
                'num_holdings': len(holdings)
            },
            'holdings': [
                {
                    'fund_name': h.fund_name,
                    'category': h.category,
                    'units': float(h.units),
                    'invested_amount': float(h.invested_amount),
                    'current_value': float(h.current_value),
                    'return_percentage': float(h.return_percentage),
                    'allocation': float((h.current_value / total_value) * 100)
                }
                for h in holdings
            ],
            'category_allocation': category_allocation_pct
        }

# portfolio/rebalancing_engine.py
from pypfopt import EfficientFrontier, risk_models, expected_returns
import pandas as pd
import numpy as np

class RebalancingEngine:
    """AI-powered portfolio rebalancing"""

    def __init__(self, model, db: Session):
        self.model = model  # Your trained Gemma/Phi-2 model
        self.db = db

    def analyze_portfolio(self, portfolio_id: int):
        """Generate rebalancing recommendations"""

        # Get portfolio holdings
        holdings = self.db.query(Holding).filter(
            Holding.portfolio_id == portfolio_id
        ).all()

        if len(holdings) == 0:
            return {"error": "Portfolio is empty"}

        # Get historical returns for each fund
        returns_data = {}
        for holding in holdings:
            historical = self.get_historical_returns(holding.scheme_code)
            returns_data[holding.fund_name] = historical

        returns_df = pd.DataFrame(returns_data)

        # Calculate expected returns using AI model
        expected_returns_dict = {}
        for holding in holdings:
            prediction = self.model.predict_fund(holding.scheme_code)
            expected_returns_dict[holding.fund_name] = prediction['expected_return'] / 100

        # Calculate risk (covariance matrix)
        cov_matrix = risk_models.sample_cov(returns_df)

        # Optimize portfolio (maximize Sharpe ratio)
        ef = EfficientFrontier(expected_returns_dict, cov_matrix)
        weights = ef.max_sharpe()
        cleaned_weights = ef.clean_weights()

        # Get current allocation
        total_value = sum([h.current_value for h in holdings])
        current_allocation = {
            h.fund_name: float(h.current_value / total_value)
            for h in holdings
        }

        # Generate rebalancing actions
        actions = []
        for fund_name, target_weight in cleaned_weights.items():
            current_weight = current_allocation.get(fund_name, 0)
            diff = target_weight - current_weight

            if abs(diff) > 0.05:  # 5% threshold
                action_type = 'BUY' if diff > 0 else 'SELL'
                amount = abs(diff) * float(total_value)

                actions.append({
                    'action': action_type,
                    'fund': fund_name,
                    'current_allocation': f'{current_weight*100:.1f}%',
                    'target_allocation': f'{target_weight*100:.1f}%',
                    'change_required': f'{abs(diff)*100:.1f}%',
                    'amount': round(amount, 2)
                })

        # Calculate expected improvement
        performance = ef.portfolio_performance()

        recommendation = {
            'portfolio_id': portfolio_id,
            'generated_at': datetime.now().isoformat(),
            'current_allocation': current_allocation,
            'recommended_allocation': cleaned_weights,
            'actions': actions,
            'expected_improvement': {
                'expected_return': f'{performance[0]*100:.2f}%',
                'volatility': f'{performance[1]*100:.2f}%',
                'sharpe_ratio': round(performance[2], 2)
            },
            'num_actions': len(actions)
        }

        # Save to database
        self.save_recommendation(recommendation)

        return recommendation
